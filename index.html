Consulta IA · Top municípios por inadimplência
SELECT
    cidade,
    COUNT(*) AS total_inadimplentes,
    ROUND(
      AVG(valor_mensalidade), 2
    ) AS ticket_medio
FROM pagamentos
WHERE status = 'INADIMPLENTE'
GROUP BY cidade
ORDER BY total_inadimplentes DESC
LIMIT 10;
